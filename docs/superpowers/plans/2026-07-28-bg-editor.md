# 배경 이미지 에디터 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 배경 슬롯 14개(스토리 7·관문 5·시스템 2)를 정의하고, 전체화면 그리드 에디터(bg.html)에서 슬롯별 이미지를 업로드하면 실제 게임 배경이 그 시점에 교체되게 한다.

**Architecture:** `src/data/backgrounds.json`이 슬롯 SSOT. `src/ui/bgSlots.ts`(순수 선택 로직) → `assets.ts`(로드)/`app.ts`(전환+페이드)가 소비. 업로드는 vite dev 플러그인(`/__bgupload`)이 `public/assets/bg/`에 기록하고 manifest를 갱신. 에디터는 flow.html 패턴의 dev 전용 페이지. 미업로드 슬롯은 기존 기본 배경으로 폴백 — 점진 적용 안전.

**Tech Stack:** TypeScript + PixiJS v8 + Vite(dev 플러그인) + Vitest. 스펙: `docs/superpowers/specs/2026-07-28-bg-editor-design.md`

- **git 레포 미생성** — 커밋 단계 없음 (Director가 레포 생성 후 일괄 커밋).

// ui/bgSlots.ts — 배경 슬롯 SSOT 접근 + 선택 규칙. 원본은 src/data/backgrounds.json (bg.html 에디터로 편집).
import backgroundsJson from "../data/backgrounds.json";

export interface BgSlot {
  id: string;
  label: string;
  file: string;
  beatIds?: string[]; // 스토리: 이 비트 도달 시 교체 (act보다 우선)
  act?: number;       // 스토리: 막 진입 시 교체
  gateId?: string;    // 관문 배경
}
interface BgManifest { story: BgSlot[]; gates: BgSlot[]; system: BgSlot[] }
export const bgManifest = backgroundsJson as unknown as BgManifest;

/** 스토리 배경 선택: beatId 일치 우선 → act 이하 최대 → null(기본 배경 폴백) */
export function pickBgSlot(beat: { id: string; act: number }, slots: BgSlot[] = bgManifest.story): BgSlot | null {
  const byBeat = slots.find((s) => s.beatIds?.includes(beat.id));
  if (byBeat) return byBeat;
  let best: BgSlot | null = null;
  for (const s of slots) {
    if (s.act === undefined || s.act > beat.act) continue;
    if (!best || (best.act ?? -1) < s.act) best = s;
  }
  return best;
}

/** 시스템 슬롯 파일 (title/loading — 없으면 null → assets.json 폴백) */
export function systemBgFile(id: string): string | null {
  return bgManifest.system.find((s) => s.id === id)?.file ?? null;
}
```

  const bgLayer = new Container();
  app.stage.addChild(bgLayer);
  stageItems.push(bgLayer);
  const fitCover = (spr: Sprite): void => {
    const s = Math.max(430 / spr.texture.width, 800 / spr.texture.height);
    spr.scale.set(s);
    spr.x = (430 - spr.texture.width * s) / 2;
    spr.y = (800 - spr.texture.height * s) / 2;
  };
  const setBg = (tex: Texture, fade: boolean): void => {
    const next = new Sprite(tex);
    fitCover(next);
    if (!fade || bgLayer.children.length === 0) {
      for (const ch of [...bgLayer.children]) ch.destroy();
      bgLayer.addChild(next);
      return;
    }
    next.alpha = 0;
    bgLayer.addChild(next);
    const t0 = performance.now();
    const step = (): void => {
      if (!alive || next.destroyed) { app.ticker.remove(step); return; }
      next.alpha = Math.min(1, (performance.now() - t0) / 300);
      if (next.alpha >= 1) {
        app.ticker.remove(step);
        for (const ch of [...bgLayer.children]) if (ch !== next) ch.destroy();
      }
    };
    app.ticker.add(step);
  };
  if (assets.background) setBg(assets.background, false);
  if (assets.background || bgManifest.story.length > 0) {
    const overlay = new Graphics().rect(0, 0, 430, 800).fill({ color: 0xf7f3fc, alpha: 0.15 });
    app.stage.addChild(overlay);
    stageItems.push(overlay);
  }
```

```ts
// 개발용: bg.html 에디터의 이미지 업로드 → public/assets/bg/ 기록 + backgrounds.json 갱신
const BG_EXTS = ['png', 'jpg', 'webp']
function bgUploadPlugin(): Plugin {
  return {
    name: 'bg-upload',
    configureServer(server) {
      server.middlewares.use('/__bgupload', (req, res) => {
        if (req.method !== 'POST') { res.statusCode = 405; res.end(); return }
        const q = new URL(req.url ?? '/', 'http://localhost').searchParams
        const slot = q.get('slot') ?? ''
        const ext = q.get('ext') ?? ''
        // slot id 형식 + 확장자 화이트리스트 — 경로는 서버가 조립(클라이언트 경로 입력 없음 → 탈출 차단)
        if (!/^[a-z0-9-]+$/.test(slot) || !BG_EXTS.includes(ext)) { res.statusCode = 400; res.end('bad params'); return }
        const manifestPath = path.resolve(process.cwd(), 'src/data/backgrounds.json')
        const manifest = JSON.parse(fs.readFileSync(manifestPath, 'utf8')) as
          Record<'story' | 'gates' | 'system', Array<{ id: string; file: string }>>
        const entry = [...manifest.story, ...manifest.gates, ...manifest.system].find((s) => s.id === slot)
        if (!entry) { res.statusCode = 400; res.end('unknown slot'); return }
        const chunks: Buffer[] = []
        let size = 0
        req.on('data', (d: Buffer) => {
          size += d.length
          if (size > 10 * 1024 * 1024) { res.statusCode = 413; res.end('too large'); req.destroy(); return }
          chunks.push(d)
        })
        req.on('end', () => {
          if (res.writableEnded) return
          const rel = `assets/bg/${slot}.${ext}`
          fs.writeFileSync(path.resolve(process.cwd(), 'public', rel), Buffer.concat(chunks))
          for (const e of BG_EXTS) { // 같은 슬롯의 다른 확장자 잔재 제거
            if (e === ext) continue
            const p = path.resolve(process.cwd(), `public/assets/bg/${slot}.${e}`)
            if (fs.existsSync(p)) fs.unlinkSync(p)
          }
          if (entry.file !== rel) { entry.file = rel; fs.writeFileSync(manifestPath, JSON.stringify(manifest, null, 2) + '\n') }
          res.end('ok')
        })
      })
    },
  }
}
```
